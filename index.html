Теплокарта взгляда

Шаг 1. Загрузите изображение:

let gazeData = []; let tracking = false; let startTime = 0; let imageElement = document.getElementById('targetImage'); let dot = document.getElementById('dot'); let status = document.getElementById('status'); let startBtn = document.getElementById('startBtn'); let calibrateBtn = document.getElementById('calibrateBtn'); let saveBtn = document.getElementById('saveBtn'); let imageInput = document.getElementById('imageInput'); // Загрузка картинки imageInput.addEventListener('change', function(e) { const file = e.target.files[0]; if (file) { const reader = new FileReader(); reader.onload = function(ev) { imageElement.src = ev.target.result; status.textContent = 'Изображение загружено.'; }; reader.readAsDataURL(file); } }); // Запуск камеры startBtn.addEventListener('click', async function() { if (!imageElement.src || imageElement.src === window.location.href) { alert('Сначала загрузите изображение!'); return; } status.textContent = 'Запуск камеры... Разрешите доступ.'; try { await webgazer.setGazeListener(function(data, elapsedTime) { if (data == null || !tracking) return; // Привязка к границам экрана, чтобы шарик не улетал let x = Math.max(0, Math.min(window.innerWidth, data.x)); let y = Math.max(0, Math.min(window.innerHeight, data.y)); // Показываем шарик на экране (фиксированно) dot.style.display = 'block'; dot.style.left = x + 'px'; dot.style.top = y + 'px'; // Сбор данных для CSV (координаты относительно страницы с учётом скролла) const rect = imageElement.getBoundingClientRect(); const imgX = x - rect.left; const imgY = y - rect.top; if (imgX >= 0 && imgX <= rect.width && imgY >= 0 && imgY <= rect.height) { const timestamp = Date.now() - startTime; gazeData.push({ x: Math.round(imgX), y: Math.round(imgY), t: timestamp }); } }).begin(); webgazer.showVideoPreview(true).showPredictionPoints(false); webgazer.applyKalmanFilter(true); tracking = true; startTime = Date.now(); gazeData = []; status.textContent = 'Камера включена. Теперь нажмите "Калибровка".'; startBtn.style.display = 'none'; calibrateBtn.style.display = 'inline-block'; saveBtn.style.display = 'inline-block'; } catch (err) { status.textContent = 'Ошибка: ' + err.message; } }); // Калибровка: 9 точек calibrateBtn.addEventListener('click', async function() { if (typeof webgazer === 'undefined') { alert('Сначала включите камеру (Шаг 2).'); return; } calibrateBtn.style.display = 'none'; status.textContent = 'Калибровка: смотрите на розовую точку и кликайте по ней.'; const margin = 80; const w = window.innerWidth; const h = window.innerHeight; const coords = [ [margin, margin], [w/2, margin], [w-margin, margin], [margin, h/2], [w/2, h/2], [w-margin, h/2], [margin, h-margin], [w/2, h-margin], [w-margin, h-margin] ]; for (let i = 0; i < coords.length; i++) { await new Promise((resolve) => { const calibDot = document.createElement('div'); calibDot.className = 'calib-dot'; calibDot.style.left = coords[i][0] + 'px'; calibDot.style.top = coords[i][1] + 'px'; document.body.appendChild(calibDot); calibDot.addEventListener('click', async (e) => { // Обучаем модель: взгляд -> клик await webgazer.recordScreenPosition(e.clientX, e.clientY, 'click'); calibDot.remove(); resolve(); }, { once: true }); }); } status.textContent = 'Калибровка завершена! Смотрите на изображение.'; calibrateBtn.style.display = 'inline-block'; }); // Сохранение CSV saveBtn.addEventListener('click', function() { tracking = false; if (typeof webgazer !== 'undefined') webgazer.pause(); if (gazeData.length === 0) { alert('Нет данных. Сначала посмотрите на изображение.'); return; } let csv = 'x,y,time_ms\n'; gazeData.forEach(function(p) { csv += p.x + ',' + p.y + ',' + p.t + '\n'; }); const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' }); const link = document.createElement('a'); link.href = URL.createObjectURL(blob); link.download = 'gaze_data_' + Date.now() + '.csv'; link.click(); status.textContent = 'Данные сохранены! Файл CSV скачан.'; });
